Campaign Management Platform

Bulk Campaign Delivery

A high-scale messaging pipeline that dispatches WhatsApp and SMS campaigns to 10,000+ recipients through batched async processing, Redis-backed delivery tracking, and intelligent SMS fallback.

The Challenge

Sending promotional messages to tens of thousands of users was not scalable — no batching, no fallback mechanism, and zero real-time visibility into delivery status.

The Solution

Built an end-to-end pipeline: Angular UI for campaign setup, Spring Boot for batched async dispatch, Redis for live delivery-state tracking, and automatic SMS fallback via Twilio.

The Impact

10,000+ recipient campaigns dispatched reliably with 96%+ delivery rates, real-time dashboard visibility, and automatic retry via SMS for unreachable WhatsApp numbers.

@Service
@RequiredArgsConstructor
public class CampaignDispatchService {

  private final WhatsAppClient                  whatsAppClient;
  private final SmsClient                       smsClient;
  private final RedisTemplate<String, String> redis;
  private final CampaignRepository              campaignRepo;

  private static final int BATCH_SIZE = 2_500;

  public void dispatch(Campaign campaign) {
    List<Recipient> recipients = campaign.getRecipients();

    // Seed Redis delivery-state for real-time dashboard tracking
    Map<String, String> statusMap = recipients.stream()
      .collect(toMap(Recipient::getId, r -> "PENDING"));
    redis.opsForHash().putAll("campaign:" + campaign.getId(), statusMap);

    // Async batched dispatch — non-blocking, respects WhatsApp rate limits
    Lists.partition(recipients, BATCH_SIZE)
         .forEach(batch ->
           CompletableFuture.runAsync(() -> processBatch(campaign, batch))
         );
  }

  private void processBatch(Campaign campaign, List<Recipient> batch) {
    batch.forEach(recipient -> {
      String msg = TemplateEngine.render(campaign.getTemplate(), recipient);
      try {
        whatsAppClient.send(recipient.getPhone(), msg);
        updateStatus(campaign.getId(), recipient.getId(), "DELIVERED");
      } catch (WhatsAppException e) {
        // Automatic SMS fallback for unreachable WhatsApp numbers
        smsClient.send(recipient.getPhone(), msg);
        updateStatus(campaign.getId(), recipient.getId(), "SMS_FALLBACK");
      }
    });
  }

  private void updateStatus(String campaignId, String recipientId, String status) {
    redis.opsForHash().put("campaign:" + campaignId, recipientId, status);
    campaignRepo.incrementCounter(campaignId, status);
  }
}
