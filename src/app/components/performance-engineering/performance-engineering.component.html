// ✅  AFTER — cache-aside: 1 Redis lookup (or 1 DB fetch, then cached)
String templateJson = redisTemplate.opsForValue().get(TEMPLATE_KEY + templateId);
if (templateJson == null) {
    MessageTemplate tpl = templateRepo.findById(templateId).orElseThrow();
    templateJson = objectMapper.writeValueAsString(tpl);
    redisTemplate.opsForValue().set(TEMPLATE_KEY + templateId, templateJson,
        1, TimeUnit.HOURS);   // TTL: 1 hour
}
recipients.forEach(r -> sendMessage(r, templateJson)); // single cache value reused
}
